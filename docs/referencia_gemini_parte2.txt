4. Utilidades: Validador RUT y Envío de QR ( backend/src/utils.js )
JavaScript
import  crypto  from   'crypto' ;
import  QRCode  from   'qrcode' ;
import  nodemailer  from   'nodemailer' ;
// Validación Módulo 11 RUT Chileno
export    function   validarRUT ( rut )  {
const  clean = rut.replace( /[^0-9kK]/g ,  '' ).toUpperCase();
if  (clean.length <  8 )  return   false ;
const  body = clean.slice( 0 , - 1 );
const  dv = clean.slice(- 1 );
let  sum =  0 ;
let  mul =  2 ;
for  ( let  i = body.length -  1 ; i >=  0 ; i--) {
sum +=  parseInt (body[i],  10 ) * mul;
mul = mul ===  7  ?  2  : mul +  1 ;
}
const  res =  11  - (sum %  11 );
const  expected = res ===  11  ?  '0'  : res ===  10  ?  'K'  : res.toString();
return  dv === expected;
}
// Formatear RUT limpio (ej: 12.345.678-9)
export    function   formatearRUT ( rut )  {
const  clean = rut.replace( /[^0-9kK]/g ,  '' ).toUpperCase();
const  body = clean.slice( 0 , - 1 );
const  dv = clean.slice(- 1 );
return   ` ${body.replace( /\B(?=(\d{3})+(?!\d))/g ,  '.' )} - ${dv} ` ;
}
// Generar firma segura para QR
export    function   generarFirmaQR ( eventId, attendeeId, rut )  {
return  crypto
.createHmac( 'sha256' , process.env.QR_SECRET_KEY)
.update( ` ${eventId} : ${attendeeId} : ${rut} ` )
.digest( 'hex' )
.slice( 0 ,  16 );
}
// Enviar correo con el nombre de la productora como remitente
export   async    function   enviarCorreoTicket ( { attendee, event, producer } )  {
const  signature = generarFirmaQR(event.id, attendee.id, attendee.rut);
const  qrPayload =  JSON .stringify({
tid : attendee.id,
eid : event.id,
sig : signature,
});
const  qrBase64 =  await  QRCode.toDataURL(qrPayload, {  width :  320 ,  margin :  2  });
const  transporter = nodemailer.createTransport({
host : process.env.SMTP_HOST,
port :  Number (process.env.SMTP_PORT),
secure :  false ,
auth : {
user : process.env.SMTP_USER,
pass : process.env.SMTP_PASS,
},
});
await  transporter.sendMail({
from :  `" ${producer.name} " < ${process.env.SMTP_FROM_EMAIL} >` ,
replyTo : producer.contact_email,
to : attendee.email,
subject :  `Tu Entrada QR -  ${event.name}  ( ${producer.name} )` ,
html :  `
<div style="font-family: sans-serif; max-width: 480px; margin: auto; background: #0f172a; color: #f8fafc; padding: 24px; border-radius: 12px;">
<h2 style="color: #22c55e; margin: 0;"> ${producer.name} </h2>

<p style="margin: 6px 0;"><strong>RUT:</strong>  ${attendee.rut} </p>
<p style="margin: 6px 0;"><strong>Entrada:</strong>  ${attendee.tier_name}  ($ ${ Number (attendee.price_paid).toLocaleString( 'es-CL' )} )</p>

5. Servidor API Principal ( backend/src/server.js )
JavaScript
import  express  from   'express' ;
import  cors  from   'cors' ;
import  { pool }  from   './db.js' ;
import  { validarRUT, formatearRUT, generarFirmaQR, enviarCorreoTicket }  from   './utils.js' ;
const  app = express();
app.use(cors());
app.use(express.json());
// 1. OBTENER ESTADO DEL EVENTO ACTIVO (Verifica si ya pasó el Popup inicial)
app.get( '/api/events/:eventId' ,  async  (req, res) => {
try  {
const  { eventId } = req.params;
const  result =  await  pool.query(
`SELECT e.*, p.name AS producer_name, p.contact_email AS producer_email
FROM events e
LEFT JOIN producers p ON e.producer_id = p.id
WHERE e.id = $1` ,
[eventId]
);
if  (result.rows.length ===  0 )  return  res.status( 404 ).json({  error :  'Evento no encontrado'  });
res.json(result.rows[ 0 ]);
}  catch  (err) {
res.status( 500 ).json({  error : err.message });
}
});
// 2. CONFIGURACIÓN ÚNICA DEL POPUP (Productora + Switch Barra)
app.post( '/api/events/:eventId/setup' ,  async  (req, res) => {
const  client =  await  pool.connect();
try  {
const  { eventId } = req.params;
const  { producerName, producerEmail, eventName, hasBar } = req.body;
await  client.query( 'BEGIN' );
// Crear o actualizar productora
const  prodRes =  await  client.query(
`INSERT INTO producers (name, contact_email) VALUES ($1, $2) RETURNING id` ,
[producerName, producerEmail]
);
const  producerId = prodRes.rows[ 0 ].id;
// Marcar evento como configurado (is_configured = TRUE) para no volver a mostrar popup
const  eventRes =  await  client.query(
`UPDATE events
SET producer_id = $1, name = COALESCE($2, name), has_bar = $3, is_configured = TRUE
WHERE id = $4 RETURNING *` ,
[producerId, eventName, hasBar, eventId]
);
await  client.query( 'COMMIT' );
res.json({ ...eventRes.rows[ 0 ],  producer_name : producerName,  producer_email : producerEmail });
}  catch  (err) {
await  client.query( 'ROLLBACK' );
res.status( 500 ).json({  error : err.message });
}  finally  {
client.release();
}
});
// 3. REGISTRAR ASISTENTE + SUMAR INGRESO + ENVIAR QR POR CORREO
app.post( '/api/events/:eventId/attendees' ,  async  (req, res) => {
const  client =  await  pool.connect();
try  {
const  { eventId } = req.params;
const  { fullName, rut, email, tierName, pricePaid } = req.body;
if  (!validarRUT(rut)) {
return  res.status( 400 ).json({  error :  'El RUT ingresado no es válido.'  });
}
const  rutFormateado = formatearRUT(rut);
await  client.query( 'BEGIN' );
// Obtener datos del evento y productora
const  evRes =  await  client.query(
`SELECT e.*, p.name AS producer_name, p.contact_email
FROM events e JOIN producers p ON e.producer_id = p.id WHERE e.id = $1` ,
[eventId]
);
const  eventData = evRes.rows[ 0 ];
// Insertar asistente
const  attRes =  await  client.query(
`INSERT INTO attendees (event_id, full_name, rut, email, price_paid, qr_token)
VALUES ($1, $2, $3, $4, $5, 'TEMP') RETURNING *` ,
[eventId, fullName, rutFormateado, email, pricePaid]
);
const  attendee = attRes.rows[ 0 ];
// Generar firma y actualizar token
const  signature = generarFirmaQR(eventId, attendee.id, rutFormateado);
await  client.query( `UPDATE attendees SET qr_token = $1 WHERE id = $2` , [signature, attendee.id]);
// Registrar automáticamente como INGRESO en finanzas
await  client.query(
`INSERT INTO transactions (event_id, type, category, description, amount)
VALUES ($1, 'INCOME', 'Entradas', $2, $3)` ,
[eventId,  `Entrada  ${tierName}  -  ${fullName}  ( ${rutFormateado} )` , pricePaid]
);
// Enviar correo con el QR y nombre de la productora
await  enviarCorreoTicket({
attendee : { ...attendee,  tier_name : tierName },
event : eventData,
producer : {  name : eventData.producer_name,  contact_email : eventData.contact_email },
});
await  client.query( 'COMMIT' );
res.json({  success :  true ,  attendee : { ...attendee,  qr_token : signature } });
}  catch  (err) {
await  client.query( 'ROLLBACK' );
res.status( 500 ).json({  error : err.message });
}  finally  {
client.release();
}
});
// 4. ESCANEAR QR EN PUERTA (Valida identidad y muestra Nombre + RUT)
app.post( '/api/events/:eventId/scan' ,  async  (req, res) => {
try  {
const  { eventId } = req.params;
const  { qrRawData } = req.body;  // JSON string leído por la cámara
const  parsed =  JSON .parse(qrRawData);
if  (parsed.eid !== eventId) {
return  res.status( 400 ).json({  status :  'INVALID' ,  message :  'QR pertenece a otro evento'  });
}
const  attRes =  await  pool.query( `SELECT * FROM attendees WHERE id = $1 AND event_id = $2` , [parsed.tid, eventId]);
if  (attRes.rows.length ===  0 ) {
return  res.status( 404 ).json({  status :  'INVALID' ,  message :  'Entrada no encontrada'  });
}
const  attendee = attRes.rows[ 0 ];
const  expectedSig = generarFirmaQR(eventId, attendee.id, attendee.rut);
if  (parsed.sig !== expectedSig) {
return  res.status( 403 ).json({  status :  'FAKE' ,  message :  'QR adulterado o inválido'  });
}
if  (attendee.is_checked_in) {
return  res.json({
status :  'ALREADY_USED' ,
message :  'ESTE QR YA INGRESÓ ANTERIORMENTE' ,
checkedInAt : attendee.checked_in_at,
attendee : {  full_name : attendee.full_name,  rut : attendee.rut },
});
}
// Marcar ingreso
await  pool.query( `UPDATE attendees SET is_checked_in = TRUE, checked_in_at = NOW() WHERE id = $1` , [attendee.id]);
res.json({
status :  'VALID' ,
message :  'ACCESO AUTORIZADO' ,
attendee : {  full_name : attendee.full_name,  rut : attendee.rut,  price_paid : attendee.price_paid },
});
}  catch  (err) {
res.status( 400 ).json({  status :  'ERROR' ,  message :  'Código QR ilegible'  });
}
});
// 5. INVENTARIO DE BARRA (Cargar stock y descontar costo en gastos)
app.post( '/api/events/:eventId/bar' ,  async  (req, res) => {
const  client =  await  pool.connect();
try  {
const  { eventId } = req.params;
const  { productName, unitCost, salePrice, initialStock } = req.body;
await  client.query( 'BEGIN' );
const  barRes =  await  client.query(
`INSERT INTO bar_inventory (event_id, product_name, unit_cost, sale_price, initial_stock, current_stock)
VALUES ($1, $2, $3, $4, $5, $5) RETURNING *` ,
[eventId, productName, unitCost, salePrice, initialStock]
);
// Sumar el costo de compra de mercadería automáticamente a GASTOS
const  totalPurchaseCost =  Number (unitCost) *  Number (initialStock);
await  client.query(
`INSERT INTO transactions (event_id, type, category, description, amount)
VALUES ($1, 'EXPENSE', 'Compra Barra', $2, $3)` ,
[eventId,  `Stock inicial:  ${initialStock} x  ${productName} ` , totalPurchaseCost]
);
await  client.query( 'COMMIT' );
res.json(barRes.rows[ 0 ]);
}  catch  (err) {
await  client.query( 'ROLLBACK' );
res.status( 500 ).json({  error : err.message });
}  finally  {
client.release();
}
});
// 6. FINANZAS: OBTENER RESUMEN Y REGISTRAR INGRESOS/GASTOS MANUALES
app.get( '/api/events/:eventId/finances' ,  async  (req, res) => {
try  {
const  { eventId } = req.params;
const  txRes =  await  pool.query( `SELECT * FROM transactions WHERE event_id = $1 ORDER BY created_at DESC` , [eventId]);
const  attRes =  await  pool.query( `SELECT * FROM attendees WHERE event_id = $1 ORDER BY created_at DESC` , [eventId]);
const  barRes =  await  pool.query( `SELECT * FROM bar_inventory WHERE event_id = $1` , [eventId]);
res.json({
transactions : txRes.rows,
attendees : attRes.rows,
barInventory : barRes.rows,
});
}  catch  (err) {
res.status( 500 ).json({  error : err.message });
}
});
app.post( '/api/events/:eventId/transactions' ,  async  (req, res) => {
try  {
const  { eventId } = req.params;
const  { type, category, description, amount } = req.body;
const  result =  await  pool.query(
`INSERT INTO transactions (event_id, type, category, description, amount) VALUES ($1, $2, $3, $4, $5) RETURNING *` ,
[eventId, type, category, description, amount]
);
res.json(result.rows[ 0 ]);
}  catch  (err) {
res.status( 500 ).json({  error : err.message });
}
});
app.listen(process.env.PORT,  () =>  {
console .log( `API Productoras corriendo en puerto  ${process.env.PORT} ` );
});

3. Escáner de Puerta ( frontend/src/components/ScannerModule.jsx )
Lee el QR con la cámara del celular y muestra en gigante el  Nombre  y  RUT  para cotejar con el carnet.
JavaScript
import  React, { useEffect, useState }  from   'react' ;
import  { Html5QrcodeScanner }  from   'html5-qrcode' ;
import  { CheckCircle2, XCircle, AlertTriangle }  from   'lucide-react' ;
export   default    function   ScannerModule ( { eventId } )  {
const  [scanResult, setScanResult] = useState( null );
useEffect( () =>  {
const  scanner =  new  Html5QrcodeScanner( 'qr-reader' , {  fps :  10 ,  qrbox :  250  });
scanner.render( async  (decodedText) => {
scanner.pause();
const  res =  await  fetch( `http://localhost:4000/api/events/ ${eventId} /scan` , {
method :  'POST' ,
headers : {  'Content-Type' :  'application/json'  },
body :  JSON .stringify({  qrRawData : decodedText }),
});
const  data =  await  res.json();
setScanResult(data);
});
return   () =>  scanner.clear().catch( () =>  {});
}, [eventId]);
const  resetScanner =  () =>  {
setScanResult( null );
window .location.reload();  // Reinicia cámara limpio
};
return  (
< div   className = "max-w-md mx-auto" >
< h2   className = "text-xl font-bold mb-4 text-center" > Control de Acceso QR </ h2 >
{!scanResult ? (
< div   id = "qr-reader"   className = "rounded-xl overflow-hidden bg-slate-900 border border-slate-800"  />
) : (
< div
className = { ` p-6   rounded-2xl   text-center   border-2  ${
scanResult.status  ===  'VALID'
? ' bg-emerald-950 / 80   border-emerald-500 '
:   scanResult.status  ===  'ALREADY_USED'
? ' bg-amber-950 / 80   border-amber-500 '
:  ' bg-red-950 / 80   border-red-500 '
}`}
>
{scanResult.status === 'VALID' &&  < CheckCircle2   className = "w-16 h-16 text-emerald-400 mx-auto mb-2"  /> }
{scanResult.status === 'ALREADY_USED' &&  < AlertTriangle   className = "w-16 h-16 text-amber-400 mx-auto mb-2"  /> }
{['INVALID', 'FAKE', 'ERROR'].includes(scanResult.status) && (
< XCircle   className = "w-16 h-16 text-red-400 mx-auto mb-2"  />
)}
< h3   className = "text-2xl font-black uppercase" > {scanResult.message} </ h3 >
{scanResult.attendee && (
< div   className = "mt-4 bg-slate-950/70 p-4 rounded-xl text-left space-y-2 border border-white/10" >
< div >
< span   className = "text-xs text-slate-400 block" > ASISTENTE: </ span >
< span   className = "text-xl font-bold text-white" > {scanResult.attendee.full_name} </ span >
</ div >
